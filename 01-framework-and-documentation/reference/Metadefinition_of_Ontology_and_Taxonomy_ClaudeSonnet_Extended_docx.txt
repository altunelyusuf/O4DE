\# Classes - Core Knowledge Organization Systems

\#################################################################

:KnowledgeOrganizationSystem rdf:type owl:Class ;

rdfs:label \"Knowledge Organization System\"@en ;

rdfs:comment \"An abstract class representing systems for organizing and
structuring knowledge\"@en ;

skos:definition \"A system for organizing information and knowledge that
may range from simple lists to complex formal ontologies\"@en .

:Ontology rdf:type owl:Class ;

rdfs:subClassOf :KnowledgeOrganizationSystem ;

rdfs:label \"Ontology\"@en ;

rdfs:comment \"A formal, explicit specification of a shared
conceptualization\"@en ;

skos:definition \"A formal representation of knowledge as a set of
concepts within a domain, with explicit specification of meaning,
machine-interpretable semantics, and shared consensus of a
community\"@en ;

skos:scopeNote \"Key characteristics: Formal, Explicit, Shared,
Machine-interpretable, Reasoning-enabled\"@en .

:Taxonomy rdf:type owl:Class ;

rdfs:subClassOf :KnowledgeOrganizationSystem ;

rdfs:label \"Taxonomy\"@en ;

rdfs:comment \"A hierarchical classification system that organizes
concepts or entities into groups based on shared characteristics\"@en ;

skos:definition \"A scheme of classification, especially a hierarchical
classification, in which things are organized into groups or types\"@en
;

owl:disjointWith :Ontology .

\# Axioms and Restrictions

\#################################################################

\# Every Ontology must be formal

:Ontology rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :isFormal ;

owl:hasValue \"true\"\^\^xsd:boolean

\] .

\# Every Ontology must be explicit

:Ontology rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :isExplicit ;

owl:hasValue \"true\"\^\^xsd:boolean

\] .

\# Every Ontology must define at least one Concept

:Ontology rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :definesConcept ;

owl:minCardinality \"1\"\^\^xsd:nonNegativeInteger

\] .

\# Every Ontology must have formal semantics

:Ontology rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :hasFormalSemantics ;

owl:someValuesFrom :FormalSemantics

\] .

\# Every Ontology must support at least one reasoning capability

:Ontology rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :supportsReasoning ;

owl:minCardinality \"1\"\^\^xsd:nonNegativeInteger

\] .

\# Every Ontology must have metadata

:Ontology rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :hasMetadata ;

owl:someValuesFrom :OntologyMetadata

\] .

\# Every Property must have a domain

:Property rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :hasDomain ;

owl:minCardinality \"1\"\^\^xsd:nonNegativeInteger

\] .

\# Every Property must have a range

:Property rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :hasRange ;

owl:minCardinality \"1\"\^\^xsd:nonNegativeInteger

\] .

\# Taxonomy must have hierarchical structure

:Taxonomy rdfs:subClassOf \[

rdf:type owl:Restriction ;

owl:onProperty :hasHierarchicalStructure ;

owl:minCardinality \"1\"\^\^xsd:nonNegativeInteger

\] .

\# Property characteristics are disjoint

\[\] rdf:type owl:AllDisjointClasses ;

owl:members ( :Transitivity :Symmetry :Functionality :Reflexivity
:Asymmetry ) .

\# Axiom types are distinct

\[\] rdf:type owl:AllDisjointClasses ;

owl:members ( :CardinalityConstraint :ValueConstraint :DisjointnessAxiom
:DomainRangeConstraint ) .

\# Comprehensive Annotations

\#################################################################

:Ontology skos:definition \"\"\"A formal, explicit specification of a
shared conceptualization where:

\- Formal: machine-readable and interpretable

\- Explicit: concepts, properties, and constraints clearly defined

\- Specification: description of concepts and relationships

\- Shared: consensus of a community

\- Conceptualization: abstract model of some phenomenon\"\"\"@en ;

skos:example \"FOAF, Dublin Core, Gene Ontology, DBpedia Ontology,
FIBO\"@en ;

skos:scopeNote \"Enables automated reasoning, knowledge sharing, and
semantic interoperability\"@en ;

skos:note \"Competency Question: What formal structures distinguish an
ontology from other KOS?\"@en ;

skos:note \"Competency Question: How does an ontology enable automated
reasoning?\"@en ;

skos:note \"Competency Question: What makes an ontology \'shared\'
versus individual?\"@en .

:Taxonomy skos:definition \"A hierarchical classification focusing
primarily on is-a relationships\"@en ;

skos:example \"Linnaean taxonomy, Dewey Decimal, NAICS codes, Amazon
product categories\"@en ;

skos:scopeNote \"Primarily hierarchical without formal logical
constraints\"@en ;

skos:note \"Competency Question: Can a taxonomy support
polyhierarchy?\"@en ;

skos:note \"Competency Question: What distinguishes a taxonomy from an
ontology?\"@en .
